    <div>
      <h1>Archivio UDL</h1>
      <p class="sub">Progettazioni per l’inclusione — infanzia, primaria, secondaria di primo grado. Uno spazio per raccogliere e creare Unità di Apprendimento progettate con i principi dell’UDL.</p>

      <div class="bar"><i id="ud-pbar"></i></div>

  </header>

  <details class="infobox" id="ud-info">
    <summary><svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.4" stroke-linecap="round" stroke-linejoin="round" aria-hidden="true"><path d="M9 6l6 6-6 6"/></svg> Che cos’è l’UDL?</summary>
    <div class="body">
      <p><b>UDL</b> è la sigla di <i>Universal Design for Learning</i> (Progettazione Universale per l’Apprendimento), il modello elaborato dal CAST (Center for Applied Special Technology, USA) per progettare fin dall’inizio percorsi accessibili alla variabilità reale di una classe, invece di adattarli in un secondo momento per chi resta escluso.</p>
      <p>Si basa su tre principi, ciascuno legato a una rete del cervello coinvolta nell’apprendimento:</p>
      <div class="principi">
        <div class="principio pr-rap"><b>Rappresentazione</b>Il «cosa» dell’apprendimento: più modi per presentare le informazioni e i contenuti.</div>
        <div class="principio pr-az"><b>Azione ed espressione</b>Il «come»: più modi per agire nell’ambiente e dimostrare ciò che si è appreso.</div>
        <div class="principio pr-co"><b>Coinvolgimento</b>Il «perché»: più modi per suscitare interesse, sostenere lo sforzo e l’autoregolazione.</div>
      </div>
      <p style="margin-top:10px">Nelle schede di questo archivio, prima dei tre principi c’è un passaggio che li prepara — le <b>possibili barriere</b>, cioè gli ostacoli che l’attività così com’è pensata potrebbe creare per qualcuno — e dopo un passaggio che li completa: le <b>attenzioni specifiche</b> per singoli alunni (compreso chi ha PEI o PDP) e le <b>evidenze per la valutazione</b>, coerenti con l’obiettivo di partenza.</p>
      <p>L’UDL non sostituisce PEI e PDP, che restano individuali: riduce il bisogno di adattamenti eccezionali perché li anticipa nella progettazione comune, per tutta la classe.</p>
      <p style="margin-bottom:0"><i>Riferimento: CAST, Universal Design for Learning Guidelines, versione 3.0, 2024 — udlguidelines.cast.org</i></p>

function matches(n,skipArea){
  var st=P(n.id);
  if(!skipArea){var ak=Object.keys(ui.areas).filter(function(k){return ui.areas[k]});if(ak.length&&ak.indexOf(n.a)<0)return false}
  if(ui.uso!=="all"&&(n.u||"").indexOf(ui.uso)<0)return false;
  if(ui.stato==="todo"&&st.k===1)return false;
  if(ui.stato==="known"&&st.k!==1)return false;
  if(ui.stato==="pin"&&!st.pin)return false;
  if(ui.q){
    var hay=norm([n.t,n.x,n.d,(n.r||[]).join(" "),n.w,n.s,areaOf(n.a).n,st.memo].join(" "));
    var toks=norm(ui.q).split(/\s+/).filter(Boolean);
    for(var i=0;i<toks.length;i++)if(hay.indexOf(toks[i])<0)return false;
  }
  return true;
}

function shuffle(a){for(var i=a.length-1;i>0;i--){var j=Math.floor(Math.random()*(i+1));var t=a[i];a[i]=a[j];a[j]=t}return a}
function startReview(includeKnown){
  rv.includeKnown=!!includeKnown;
  var v=visible().filter(function(n){return rv.includeKnown||P(n.id).k!==1}).map(function(n){return n.id});
  rv.deck=shuffle(v);rv.i=0;rv.shown=false;rv.known=0;rv.again=0;
  var d=document.getElementById("cn-dRev");if(!d.open)d.showModal();
  renderReview();
}
function renderReview(){
  var sh=document.getElementById("cn-rSheet");
  if(!rv.deck.length){
    sh.className="sheet plain";
    sh.innerHTML='<button class="icon-btn close" data-act="closeR" aria-label="Chiudi">'+IC.x+'</button><h2>Niente da ripassare</h2><p>Con i filtri attuali tutte le note sono già memorizzate.     return;
  }
  if(rv.i>=rv.deck.length){
    sh.className="sheet plain";
    sh.innerHTML='<button class="icon-btn close" data-act="closeR" aria-label="Chiudi">'+IC.x+'</button><h2>Ripasso completato</h2><p><b>'+rv.known+'</b> note segnate come memorizzate, <b>'+rv.again+'</b> da rivedere.</p><div class="actions"><button class="btn primary" data-act="revagain">Ripassa di nuovo</button><button class="btn" data-act="closeR">Torna alla bacheca</button></div>';

  }

}
